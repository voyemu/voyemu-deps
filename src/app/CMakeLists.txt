# Release optimizes for size: outside the per-frame window path the launcher waits on I/O, Java and the
# guests. Sources that touch every pixel or byte of a frame take -O2 (voyemu_frame_path_sources and the
# SDL3 pixel sources below). Own sections let ld.lld and lld-link drop unreferenced code and data; the
# Mach-O linker drops them per symbol. See docs/dev/build.md.
foreach(lang C CXX OBJC OBJCXX)
    set(CMAKE_${lang}_FLAGS_RELEASE "-Os -DNDEBUG")
endforeach()
add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX,OBJC,OBJCXX>:-ffunction-sections;-fdata-sections>")
# The C sources are the pinned SDL3, curl, nghttp2 and miniz, and no exception unwinds through them:
# the callbacks they call are noexcept or catch. On Linux they carry no unwind tables; Windows x64
# requires them and the Mach-O linker keeps its compact unwind.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_compile_options("$<$<COMPILE_LANGUAGE:C>:-fno-asynchronous-unwind-tables;-fno-unwind-tables>")
endif()

function(voyemu_test_source name output)
    file(GLOB matches "${CMAKE_CURRENT_SOURCE_DIR}/*/${name}.test.cpp")
    list(LENGTH matches count)
    if(NOT count EQUAL 1)
        message(FATAL_ERROR "Expected one adjacent test source: ${name}")
    endif()
    set(${output} "${matches}" PARENT_SCOPE)
endfunction()
# The only test registration: voyemu_add_test(<name> LABELS <label>... TIMEOUT <seconds> [RUN_SERIAL]
# COMMAND <command>...). Labels: host needs only the build tree and the pinned build tools, deps needs
# toolchains installed by deps, oem needs imported OEM inputs. Exit code 77 reports absent inputs as skipped.
function(voyemu_add_test name)
    cmake_parse_arguments(PARSE_ARGV 1 arg "RUN_SERIAL" "TIMEOUT" "LABELS;COMMAND")
    if(NOT arg_LABELS OR NOT arg_TIMEOUT OR NOT arg_COMMAND OR arg_UNPARSED_ARGUMENTS)
        message(FATAL_ERROR "voyemu_add_test(${name}) requires LABELS, TIMEOUT and COMMAND")
    endif()
    foreach(label IN LISTS arg_LABELS)
        if(NOT label MATCHES "^(host|deps|oem|source-fixture)$")
            message(FATAL_ERROR "voyemu_add_test(${name}): unknown label ${label}")
        endif()
    endforeach()
    add_test(NAME ${name} COMMAND ${arg_COMMAND})
    # Relative test scratch paths belong to the project, never the candidate executable directory.
    set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${voyemu_project_root}" LABELS "${arg_LABELS}" TIMEOUT ${arg_TIMEOUT} SKIP_RETURN_CODE 77 RUN_SERIAL ${arg_RUN_SERIAL})
endfunction()

        set_target_properties(${name}_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
        voyemu_add_test(${name} LABELS host TIMEOUT 60 COMMAND ${name}_test)

    set(voyemu_cjk_texts "${CMAKE_CURRENT_SOURCE_DIR}/ui/ui_localization.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/cli/messages.zh.json")
    set(voyemu_cjk_inputs "")
    foreach(input IN ITEMS "${voyemu_cjk_subsetter}" "${voyemu_cjk_font}" ${voyemu_cjk_texts})
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input}")
        file(SHA256 "${input}" digest)
        string(APPEND voyemu_cjk_inputs "${digest}\n")
    endforeach()
    foreach(weight IN ITEMS 400 500 700)
        set(voyemu_cjk_output "${CMAKE_CURRENT_BINARY_DIR}/noto-sans-sc-${weight}.ttf")
        set(voyemu_cjk_previous "")
        if(EXISTS "${voyemu_cjk_output}.inputs")
            file(READ "${voyemu_cjk_output}.inputs" voyemu_cjk_previous)
        endif()
        if(NOT EXISTS "${voyemu_cjk_output}" OR NOT voyemu_cjk_previous STREQUAL voyemu_cjk_inputs)
            execute_process(COMMAND "${voyemu_jdk}/bin/java" "${voyemu_cjk_subsetter}" "${voyemu_cjk_font}" ${weight} "${voyemu_cjk_output}" ${voyemu_cjk_texts} RESULT_VARIABLE voyemu_result OUTPUT_VARIABLE voyemu_output ERROR_VARIABLE voyemu_output)
            if(NOT voyemu_result EQUAL 0)
                message(FATAL_ERROR "Chinese UI font subset failed: ${voyemu_output}")
            endif()
            file(WRITE "${voyemu_cjk_output}.inputs" "${voyemu_cjk_inputs}")
        endif()
    endforeach()

    foreach(pair IN ITEMS "CLUSTER_UI_FONT|${CLUSTER_FONT}|inter-regular" "CLUSTER_UI_TAB_FONT|${CLUSTER_TAB_FONT}|inter-medium" "CLUSTER_UI_BOLD_FONT|${CLUSTER_BOLD_FONT}|inter-bold")
        string(REPLACE "|" ";" entry "${pair}")
        list(GET entry 0 variable)
        list(GET entry 1 source)
        list(GET entry 2 name)
        set(${variable} "${CMAKE_CURRENT_BINARY_DIR}/${name}.ttf")
        set(voyemu_ui_font_inputs "")
        foreach(input IN ITEMS "${voyemu_cjk_subsetter}" "${source}" ${voyemu_ui_texts})

            file(SHA256 "${input}" digest)
            string(APPEND voyemu_ui_font_inputs "${digest}\n")
        endforeach()
        set(voyemu_ui_font_previous "")
        if(EXISTS "${${variable}}.inputs")
            file(READ "${${variable}}.inputs" voyemu_ui_font_previous)
        endif()
        if(NOT EXISTS "${${variable}}" OR NOT voyemu_ui_font_previous STREQUAL voyemu_ui_font_inputs)
            execute_process(COMMAND "${voyemu_jdk}/bin/java" "${voyemu_cjk_subsetter}" --static "${source}" "${${variable}}" ${voyemu_ui_texts} RESULT_VARIABLE voyemu_result OUTPUT_VARIABLE voyemu_output ERROR_VARIABLE voyemu_output)
            if(NOT voyemu_result EQUAL 0)
                message(FATAL_ERROR "UI font subset failed: ${voyemu_output}")
            endif()
            file(WRITE "${${variable}}.inputs" "${voyemu_ui_font_inputs}")
        endif()
    endforeach()
    set(IMGUI "${VOYEMU_VENDOR_ROOT}/imgui")
    if(APPLE)

        set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    endif()
    include("${CMAKE_CURRENT_SOURCE_DIR}/window_dependencies.cmake")

    # their pixels to RGBA. SDL3's precompiled header is built with -Os, so these sources skip it.
    get_target_property(voyemu_sdl_frame_sources SDL3-static SOURCES)
    list(FILTER voyemu_sdl_frame_sources INCLUDE REGEX "/src/video/(SDL_blit[^/]*|SDL_pixels|SDL_stb|SDL_surface)\\.c$")
    set_source_files_properties(${voyemu_sdl_frame_sources} TARGET_DIRECTORY SDL3-static PROPERTIES COMPILE_OPTIONS -O2 SKIP_PRECOMPILE_HEADERS ON)
    add_library(voyemu_imgui STATIC "${IMGUI}/imgui.cpp" "${IMGUI}/imgui_draw.cpp" "${IMGUI}/imgui_tables.cpp" "${IMGUI}/imgui_widgets.cpp")

    target_sources(voyemu PRIVATE platform/window_sdl.cpp "${IMGUI}/backends/imgui_impl_sdl3.cpp" "${IMGUI}/backends/imgui_impl_sdlrenderer3.cpp")
    target_compile_definitions(voyemu PRIVATE VOYEMU_HAS_WINDOW=1)
